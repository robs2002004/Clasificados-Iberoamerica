rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function estaAutenticado() {
      return request.auth != null;
    }

    function esDueno(uid) {
      return estaAutenticado() && request.auth.uid == uid;
    }

    function esAdmin() {
      return estaAutenticado() &&
        exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    // ---- Colección: anuncios ----
    match /anuncios/{anuncioId} {

      // Cualquiera puede leer anuncios activos.
      // El dueño puede leer los suyos aunque estén vencidos o pendientes de pago.
      // El admin puede leer todo.
      allow read: if resource.data.estado == 'activo'
                  || esDueno(resource.data.uid)
                  || esAdmin();

      // Crear: solo un usuario autenticado, y solo a nombre de sí mismo.
      allow create: if estaAutenticado()
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.titulo is string
                    && request.resource.data.titulo.size() > 0
                    && request.resource.data.whatsapp is string
                    // El plan básico se activa directo; los planes pagados
                    // deben crearse como "pendiente" (los confirma la Cloud Function).
                    && (
                         (request.resource.data.plan == 'basico'
                           && request.resource.data.estadoPago == 'gratis'
                           && request.resource.data.estado == 'activo')
                         ||
                         (request.resource.data.plan != 'basico'
                           && request.resource.data.estadoPago == 'pendiente'
                           && request.resource.data.estado == 'pendiente')
                       );

      // Actualizar / eliminar: solo el dueño del anuncio o el admin.
      // El campo estadoPago NUNCA se puede modificar desde el cliente
      // (solo la Cloud Function, que usa el Admin SDK y salta estas reglas).
      allow update: if (esDueno(resource.data.uid) || esAdmin())
                    && request.resource.data.estadoPago == resource.data.estadoPago;

      allow delete: if esDueno(resource.data.uid) || esAdmin();
    }

    // ---- Colección: admins (solo lectura desde reglas, se administra manualmente) ----
    match /admins/{uid} {
      allow read: if estaAutenticado();
      allow write: if false;
    }
  }
}
